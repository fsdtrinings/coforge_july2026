#use of having :  i want product details where average rating is more than 4
SELECT ProductID, ROUND(AVG(Rating), 1) AS AvgRating
FROM Feedback
GROUP BY ProductID
HAVING AVG(Rating) >= 4;

#or products with more than 2 feedbacks or 5 feedbacks
SELECT ProductID, COUNT(*) AS FeedbackCount
FROM Feedback
GROUP BY ProductID
HAVING COUNT(*) > 2;

# All orders of Laptop 
# so i want to check all the orders placed by whom (what is the customerId) and on which date they placed related to Product category
# lets say during discount week 
# all order infomation is inside productorder table , match orderid from product order and orderItem and get OrderItems only for laptops
Select po.orderid,po.customerid,po.orderdate , 'laptop' as category from productorder as po
where orderid in (
		select OrderID from 
        orderItem 
        where productId in 
        ( select productid from product where category = 'laptop')
        );
