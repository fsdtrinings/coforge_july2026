# inner Join with where clause with multiple Joins 
# we want OrderId & Status , ProductName & category  & Quantity in each Order  which is available in order item

select po.orderId , po.status , p.productName , p.category , oi.quantity 
from productorder po
Join orderitem oi
	on oi.orderID = po.OrderID
Join product p 
	on p.ProductID = oi.ProductID
where Po.Status = 'Confirmed' 
and
	  p.category = 'Food';

# -----------------------------------------------------------------------------------------------------------
#  left join  : Put the table in left which comes first in the from clausee
# Even if the result is the same, LEFT JOIN is more readable and predictable 
#  Developers expect the “main” table to come first.
# -----------------------------------------------------------------------------------------------------------
#  So Left Join includes all rows of Left table and keep right table column values null and vice cersa

SELECT o.OrderID, s.Status
FROM productOrder o
LEFT JOIN Shipment s ON o.OrderID = s.OrderID;

# lets compare the feedback of Same product with different feedback
SELECT 
  f1.FeedbackID AS FeedbackIDA,
  f2.FeedbackID AS FeedbackIDB,
  f1.ProductID,
  p.ProductName,
  p.Category,
  f1.Rating AS RatingA,
  f2.Rating AS RatingB
FROM Feedback f1
JOIN Feedback f2 
  ON f1.ProductID = f2.ProductID AND f1.FeedbackID <> f2.FeedbackID
JOIN Product p 
  ON f1.ProductID = p.ProductID;
  
  # or getting the Products of Similar Category 
  SeLECT

  p2.ProductName AS ProductB_Name,

# Very Important 
# I want to See the product having Max orders in its Categorty
  SELECT 
  p.ProductID,
  p.ProductName,
  p.Category,
  COUNT(oi.OrderID) AS TotalOrders
FROM Product p
JOIN OrderItem oi ON p.ProductID = oi.ProductID
GROUP BY p.ProductID, p.ProductName, p.Category
HAVING COUNT(oi.OrderID) = (
  SELECT MAX(OrderCount)
  FROM (
    SELECT COUNT(*) AS OrderCount
    FROM Product p2
    JOIN OrderItem oi2 ON p2.ProductID = oi2.ProductID
    WHERE p2.Category = p.Category
    GROUP BY p2.ProductID
  ) AS CategoryOrders
);
  
 # and Applying orderBy 
SELECT 
  p.ProductID,
  p.ProductName,
  p.Category,
  COUNT(oi.OrderID) AS TotalOrders
FROM Product p
JOIN OrderItem oi 
	ON p.ProductID = oi.ProductID
GROUP BY p.ProductID, p.ProductName, p.Category
HAVING COUNT(oi.OrderID) = (
  SELECT MAX(OrderCount)
  FROM (
    SELECT COUNT(*) AS OrderCount
    FROM Product p2
    JOIN OrderItem oi2 ON p2.ProductID = oi2.ProductID
    WHERE p2.Category = p.Category
    GROUP BY p2.ProductID
  ) AS CategoryOrders
)
ORDER BY TotalOrders DESC;
